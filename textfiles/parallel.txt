nano parallel_sum.c

#include <stdio.h>
#include <mpi.h>

int main(int argc, char *argv[]) {
    int rank, size;
    int local_sum, total_sum;
    int start, end, i;

    MPI_Init(&argc, &argv);

    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    start = rank * 10 + 1;
    end = start + 9;

    local_sum = 0;

    for (i = start; i <= end; i++) {
        local_sum += i;
    }

    MPI_Reduce(&local_sum, &total_sum, 1,
               MPI_INT, MPI_SUM, 0, MPI_COMM_WORLD);

    if (rank == 0) {
        printf("Total sum = %d\n", total_sum);
    }

    MPI_Finalize();
    return 0;
}

nano omp_array_sum.c



#include <stdio.h>
#include <omp.h>

int main() {
    int a[100];
    int i;
    int sum = 0;

    // CHANGE: Initialize the array with values 1 to 100
    for (i = 0; i < 100; i++) {
        a[i] = i + 1;
    }

    // COMMON: Parallelize the loop and combine partial sums
    #pragma omp parallel for reduction(+:sum)
    for (i = 0; i < 100; i++) {
        sum += a[i];
    }

    // COMMON: Print the final result once
    printf("Sum of 100 elements = %d\n", sum);

    return 0;
}

#include <stdio.h>
#include <mpi.h>

int main(int argc, char *argv[]) {
    int rank, size;
    int a[8] = {12, 45, 3, 89, 21, 6, 54, 10};
    int local_max, local_min;
    int max, min, i;

    MPI_Init(&argc, &argv);

    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    local_max = a[rank * 2];
    local_min = a[rank * 2];

    for (i = rank * 2; i < rank * 2 + 2; i++) {
        if (a[i] > local_max)
            local_max = a[i];

        if (a[i] < local_min)
            local_min = a[i];
    }

    MPI_Reduce(&local_max, &max, 1, MPI_INT,
               MPI_MAX, 0, MPI_COMM_WORLD);

    MPI_Reduce(&local_min, &min, 1, MPI_INT,
               MPI_MIN, 0, MPI_COMM_WORLD);

    if (rank == 0) {
        printf("Maximum = %d\n", max);
        printf("Minimum = %d\n", min);
    }

    MPI_Finalize();
    return 0;
}

#include <stdio.h>
#include <string.h>
#include <mpi.h>

int main(int argc, char *argv[]) {
    char words[8][20] = {
        "apple", "mango", "apple", "grape",
        "orange", "mango", "kiwi", "banana"
    };

    int rank, size, i, j, count;
    int unique[8] = {0};

    MPI_Init(&argc, &argv);
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    for (i = rank; i < 8; i += size) {
        count = 0;

        for (j = 0; j < 8; j++) {
            if (strcmp(words[i], words[j]) == 0)
                count++;
        }

        if (count == 1)
            unique[i] = 1;
    }

    int all_unique[8];

    MPI_Reduce(unique, all_unique, 8, MPI_INT,
               MPI_MAX, 0, MPI_COMM_WORLD);

    if (rank == 0) {
        printf("Unique words:\n");

        for (i = 0; i < 8; i++) {
            if (all_unique[i] == 1)
                printf("%s\n", words[i]);
        }
    }

    MPI_Finalize();
    return 0;
}
